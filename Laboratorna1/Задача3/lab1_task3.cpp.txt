#include <iostream>

using namespace std;

int main() {
    int X = 5;
    int Y = 10;
    int Z;
    
    cout << "--- Задание 3: Инкрементация и Декрементация ---" << endl;
    
    // --- Постфиксная инкрементация (X++) ---
    cout << "\nНачальное значение X: " << X << endl; // X = 5
    
    // Z присваивается старое значение X (5), затем X увеличивается до 6
    Z = X++;
    cout << "Результат Z = X++: Z = " << Z << endl; // Z = 5
    cout << "Новое значение X: " << X << endl;     // X = 6
    
    // --- Префиксная инкрементация (++Y) ---
    cout << "\nНачальное значение Y: " << Y << endl; // Y = 10

    // Y сначала увеличивается до 11, затем Z присваивается новое значение Y (11)
    Z = ++Y;
    cout << "Результат Z = ++Y: Z = " << Z << endl; // Z = 11
    cout << "Новое значение Y: " << Y << endl;     // Y = 11
    
    // --- Декрементация (Y--) ---
    cout << "\nНачальное значение Y: " << Y << endl; // Y = 11
    Y--; // Эквивалентно Y = Y - 1
    cout << "Y после декрементации (Y--): " << Y << endl; // Y = 10

    return 0;
}
