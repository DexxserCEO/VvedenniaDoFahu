#include <iostream>

using namespace std;

int main() {
    int grade;
    
    cout << "--- Задание 5: Оценка студента ---" << endl;
    
    // Ввод оценки
    cout << "Введите оценку студента (от 0 до 100): ";
    cin >> grade;

    // Проверка, находится ли оценка в допустимом диапазоне
    if (grade < 0 || grade > 100) {
        cout << "Ошибка: Введена недопустимая оценка. Оценка должна быть от 0 до 100." << endl;
    } 
    // Используем конструкцию if-else if для шкалы
    else if (grade >= 90) {
        cout << "Ваша оценка: 'отлично'" << endl;
    } else if (grade >= 75) {
        cout << "Ваша оценка: 'хорошо'" << endl;
    } else if (grade >= 60) {
        cout << "Ваша оценка: 'удовлетворительно'" << endl;
    } else {
        // Все, что ниже 60 (но выше или равно 0)
        cout << "Ваша оценка: 'неудовлетворительно'" << endl;
    }

    return 0;
}
